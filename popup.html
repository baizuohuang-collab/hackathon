// ContextShift Local AI Engine & Controller document.addEventListener('DOMContentLoaded', () => { const summarizeBtn = document.getElementById('summarizeBtn'); const clusterBtn = document.getElementById('clusterBtn'); const extractActionBtn = document.getElementById('extractActionBtn'); const outputDiv = document.getElementById('output'); const providerSelect = document.getElementById('providerSelect'); const apiKeyInput = document.getElementById('apiKeyInput'); const saveSettingsBtn = document.getElementById('saveSettingsBtn'); // Load user settings chrome.storage.local.get(['aiProvider', 'apiKey'], (data) => { if (data.aiProvider) providerSelect.value = data.aiProvider; if (data.apiKey) apiKeyInput.value = data.apiKey; }); // Save Settings saveSettingsBtn.addEventListener('click', () => { chrome.storage.local.set({ aiProvider: providerSelect.value, apiKey: apiKeyInput.value.trim() }, () => { showNotification('Settings saved successfully!'); }); }); // 1-Click Summarize Current Tab summarizeBtn.addEventListener('click', async () => { renderLoading('Analyzing page content...'); const [tab] = await chrome.tabs.query({ active: true, currentWindow: true }); chrome.scripting.executeScript({ target: { tabId: tab.id }, func: extractPageText }, async (results) => { if (!results || !results[0] || !results[0].result) { renderError('Failed to extract content from tab.'); return; } const pageText = results[0].result; const provider = providerSelect.value; const apiKey = apiKeyInput.value.trim(); if (provider === 'local' || !apiKey) { // Use Local AI Processing Engine (0 API Key Required) const summary = runLocalSummarizer(pageText, tab.title); renderSummary(tab.title, summary, 'Local Offline AI'); } else { // Use External API Call (Gemini / OpenAI) try { const summary = await callExternalAI(provider, apiKey, pageText, 'Summarize this webpage'); renderSummary(tab.title, summary, provider.toUpperCase()); } catch (err) { // Fallback gracefully to Local AI if API fails const summary = runLocalSummarizer(pageText, tab.title); renderSummary(tab.title, summary, 'Local AI (Fallback - API Error)'); } } }); }); // Helper: Content Extraction Function function extractPageText() { const paragraphs = Array.from(document.querySelectorAll('p, h1, h2, h3, li')); return paragraphs.map(p => p.innerText.trim()).filter(t => t.length > 20).join('\n').substring(0, 8000); } // Local JS NLP Engine (No API Key Needed!) function runLocalSummarizer(text, title) { if (!text || text.length < 50) return ["Short page content. Not enough text to build summary."]; const sentences = text.split(/(?<=[.?!])\s+/); // Tokenize & Frequency Scoring const wordFreq = {}; const stopWords = new Set(['the', 'is', 'at', 'which', 'on', 'and', 'a', 'an', 'in', 'to', 'for', 'of', 'with', 'this', 'that', 'it', 'as', 'are', 'be', 'by', 'or', 'from']); const words = text.toLowerCase().match(/\b[a-z]{3,}\b/g) || []; words.forEach(word => { if (!stopWords.has(word)) { wordFreq[word] = (wordFreq[word] || 0) + 1; } }); // Score sentences based on key terms const scoredSentences = sentences.map(sentence => { let score = 0; const sentenceWords = sentence.toLowerCase().match(/\b[a-z]{3,}\b/g) || []; sentenceWords.forEach(w => { if (wordFreq[w]) score += wordFreq[w]; }); return { sentence: sentence.trim(), score: score / (sentenceWords.length || 1) }; }); // Sort and select top key points scoredSentences.sort((a, b) => b.score - a.score); const topPoints = scoredSentences.slice(0, 4).map(s => s.sentence); return topPoints; } // Render Helpers function renderLoading(msg) { outputDiv.innerHTML = `
${msg}
`; } function renderSummary(title, points, modelName) { let html = `
${modelName}

${escapeHtml(title)}

    `; points.forEach(pt => { html += `
  • ${escapeHtml(pt)}
    • `; }); html += `
`; outputDiv.innerHTML = html; } function renderError(msg) { outputDiv.innerHTML = `
${msg}
`; } function showNotification(msg) { const notif = document.createElement('div'); notif.className = 'notif'; notif.innerText = msg; document.body.appendChild(notif); setTimeout(() => notif.remove(), 2500); } function escapeHtml(str) { return str.replace(/&/g, "&").replace(//g, ">"); } });
